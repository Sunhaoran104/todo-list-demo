待办清单

const input = document.getElementById('task-input'); const addBtn = document.getElementById('add-btn'); const taskList = document.getElementById('task-list'); function loadTask() { const taskData = JSON.parse(localStorage.getItem('todoTask')) || []; taskList.innerHTML = ''; taskData.forEach((item, index) => createTaskDom(item, index)); } function createTaskDom(taskInfo, index) { const li = document.createElement('li'); li.className = taskInfo.finish ? 'finish' : ''; li.innerHTML = ` ${taskInfo.text} `; // 勾选完成事件 li.querySelector('input').onchange = function () { let taskArr = JSON.parse(localStorage.getItem('todoTask')); taskArr[index].finish = this.checked; localStorage.setItem('todoTask', JSON.stringify(taskArr)); loadTask(); }; // 删除事件 li.querySelector('.del').onclick = function () { let taskArr = JSON.parse(localStorage.getItem('todoTask')); taskArr.splice(index, 1); localStorage.setItem('todoTask', JSON.stringify(taskArr)); loadTask(); }; taskList.appendChild(li); } function saveTask() { const allTask = []; document.querySelectorAll('#task-list li').forEach(li => { allTask.push({ text: li.querySelector('span').innerText, finish: li.querySelector('input').checked }) }) localStorage.setItem('todoTask', JSON.stringify(allTask)); } addBtn.onclick = function () { const text = input.value.trim(); if (!text) { alert('不能提交空白任务！'); return; } createTaskDom({ text, finish: false }); saveTask(); input.value = ''; } window.onload = loadTask;
