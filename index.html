const input = document.getElementById('taskInput'); const addBtn = document.getElementById('addBtn'); const taskList = document.getElementById('taskList'); function loadTask() { const taskData = JSON.parse(localStorage.getItem('todoTask')) || []; taskList.innerHTML = ''; taskData.forEach(item => createTaskDom(item)); } function createTaskDom(taskInfo) { const li = document.createElement('li'); li.className = taskInfo.finish ? 'finish' : ''; li.innerHTML = ` ${taskInfo.text} `; li.querySelector('input').onchange = function () { taskInfo.finish = this.checked; saveTask(); loadTask(); }; / li.querySelector('.del').onclick = function () { const taskArr = JSON.parse(localStorage.getItem('todoTask')); const index = taskArr.findIndex(t => t.text === taskInfo.text); taskArr.splice(index, 1); localStorage.setItem('todoTask', JSON.stringify(taskArr)); loadTask(); }; taskList.appendChild(li); } function saveTask() { const allTask = []; document.querySelectorAll('#taskList li').forEach(li => { allTask.push({ text: li.querySelector('span').innerText, finish: li.querySelector('input').checked }) }) localStorage.setItem('todoTask', JSON.stringify(allTask)); } addBtn.onclick = function () { const text = input.value.trim(); if (!text) { alert('不能提交空白任务！'); return; } createTaskDom({ text, finish: false }); saveTask(); input.value = ''; } window.onload = loadTask;
